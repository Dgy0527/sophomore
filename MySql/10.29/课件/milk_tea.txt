-- 创建数据库
CREATE DATABASE IF NOT EXISTS milk_tea DEFAULT CHARACTER SET utf8mb4;
USE milk_tea;
-- 1. 奶茶商品表 goods
DROP TABLE IF EXISTS goods;

CREATE TABLE goods
(
gid INT PRIMARY KEY AUTO_INCREMENT COMMENT '商品ID',
gname VARCHAR(50) NOT NULL COMMENT '奶茶名称',
price DECIMAL(5,2) COMMENT '单价',
category VARCHAR(20)
)ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 2. 订单表 orders
DROP TABLE IF EXISTS orders;
CREATE TABLE orders
(   
oid INT PRIMARY KEY AUTO_INCREMENT COMMENT '订单ID',
gid INT COMMENT '关联商品ID',   
sname VARCHAR(20) COMMENT '下单学生昵称',    
num INT COMMENT '购买数量',    
otime DATETIME COMMENT '下单时间'
)ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
